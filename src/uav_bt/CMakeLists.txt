cmake_minimum_required(VERSION 3.16)
project(uav_bt)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(uav_nav_interfaces REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(mission_runner src/mission_runner.cpp)
target_compile_features(mission_runner PRIVATE cxx_std_17)
target_compile_options(mission_runner PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(mission_runner rclcpp rclcpp_action ament_index_cpp uav_nav_interfaces)
target_link_libraries(mission_runner behaviortree_cpp::behaviortree_cpp nlohmann_json::nlohmann_json)
add_executable(algorithm_mission_server src/algorithm_mission_server.cpp)
target_compile_features(algorithm_mission_server PRIVATE cxx_std_17)
target_compile_options(algorithm_mission_server PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(algorithm_mission_server rclcpp rclcpp_action rcl_interfaces ament_index_cpp uav_nav_interfaces)
target_link_libraries(algorithm_mission_server behaviortree_cpp::behaviortree_cpp nlohmann_json::nlohmann_json OpenSSL::Crypto)
install(TARGETS mission_runner algorithm_mission_server DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY trees DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(bt_algorithm_contract test/test_algorithm_mission.py TIMEOUT 120
    ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_pytest_test(bt_ros_contract test/test_runner_integration.py TIMEOUT 90
    ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
endif()
ament_package()
