cmake_minimum_required(VERSION 3.16)
project(uav_ego_adapter)
find_package(ament_cmake REQUIRED)
find_package(ego_planner_vendor REQUIRED)
find_package(uav_nav_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(ego_nvblox_planner src/planner.cpp)
target_compile_options(ego_nvblox_planner PRIVATE -O2)
target_compile_features(ego_nvblox_planner PRIVATE cxx_std_17)
target_link_libraries(ego_nvblox_planner ego_planner_vendor::ego_core)
ament_target_dependencies(ego_nvblox_planner rclcpp nav_msgs geometry_msgs std_msgs uav_nav_interfaces)
install(TARGETS ego_nvblox_planner DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_free_volume test/test_free_volume.cpp)
  target_link_libraries(test_free_volume ego_planner_vendor::ego_core)
  target_compile_features(test_free_volume PRIVATE cxx_std_17)
  ament_add_gtest(test_safe_seed test/test_safe_seed.cpp)
  target_link_libraries(test_safe_seed ego_planner_vendor::ego_core)
  target_compile_features(test_safe_seed PRIVATE cxx_std_17)
  ament_add_gtest(test_astar_endpoints test/test_astar_endpoints.cpp)
  target_link_libraries(test_astar_endpoints ego_planner_vendor::ego_core)
  target_compile_features(test_astar_endpoints PRIVATE cxx_std_17)
endif()
ament_package()
