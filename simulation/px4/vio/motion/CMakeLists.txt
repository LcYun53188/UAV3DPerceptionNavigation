cmake_minimum_required(VERSION 3.16)
project(vio_motion_carrier LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(vio_motion_carrier SHARED MotionCarrier.cc)
target_compile_features(vio_motion_carrier PRIVATE cxx_std_17)
target_compile_options(vio_motion_carrier PRIVATE -Wall -Wextra -Werror)
target_link_libraries(vio_motion_carrier PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
