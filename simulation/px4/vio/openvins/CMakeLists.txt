cmake_minimum_required(VERSION 3.16)
project(uav_openvins_replay)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
set(CMAKE_CXX_STANDARD 17)
add_executable(openvins_replay ../../../../scripts/simulation/vio/sdk/openvins_covariance_replay.cpp)
target_compile_definitions(openvins_replay PRIVATE ROS_AVAILABLE=2)
target_include_directories(openvins_replay PRIVATE ${OpenCV_INCLUDE_DIRS} ../../../../scripts/simulation/vio/sdk)
ament_target_dependencies(openvins_replay rclcpp ov_core ov_init ov_msckf)
target_link_libraries(openvins_replay Eigen3::Eigen ${OpenCV_LIBRARIES} Boost::filesystem Boost::system)
add_executable(openvins_node ../../../../scripts/simulation/vio/sdk/openvins_ros_node.cpp)
target_compile_definitions(openvins_node PRIVATE ROS_AVAILABLE=2)
target_include_directories(openvins_node PRIVATE ${OpenCV_INCLUDE_DIRS} ../../../../scripts/simulation/vio/sdk)
ament_target_dependencies(openvins_node rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs ov_core ov_init ov_msckf)
target_link_libraries(openvins_node Eigen3::Eigen ${OpenCV_LIBRARIES} Boost::filesystem Boost::system)
