cmake_minimum_required(VERSION 3.16)
project(vio_same_aircraft_fixture LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(vio_same_aircraft_fixture SHARED SameAircraftMotion.cc)
target_compile_features(vio_same_aircraft_fixture PRIVATE cxx_std_17)
target_compile_options(vio_same_aircraft_fixture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(vio_same_aircraft_fixture PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core)
